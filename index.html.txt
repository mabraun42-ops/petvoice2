    <script>
        // Sprachumschaltung
        function setLanguage(lang) {
            const deElements = document.querySelectorAll('.lang-de');
            const jaElements = document.querySelectorAll('.lang-ja');
            const btnDe = document.getElementById('btn-de');
            const btnJa = document.getElementById('btn-ja');
            if (lang === 'ja') {
                deElements.forEach(el => el.style.display = 'none');
                jaElements.forEach(el => el.style.display = 'block');
                btnJa.classList.add('bg-white', 'shadow-sm', 'text-indigo-600');
                btnDe.classList.remove('bg-white', 'shadow-sm', 'text-indigo-600');
            } else {
                jaElements.forEach(el => el.style.display = 'none');
                deElements.forEach(el => el.style.display = 'block');
                btnDe.classList.add('bg-white', 'shadow-sm', 'text-indigo-600');
                btnJa.classList.remove('bg-white', 'shadow-sm', 'text-indigo-600');
            }
        }

        // NEU: Formular-Logik mit KOSTENLOSER Weiterleitung
        const orderForm = document.getElementById('orderForm');
        const loadingOverlay = document.getElementById('loading-overlay');
        const stripeLink = "https://buy.stripe.com/test_6oU5kDg5E5L2ckg4Do4Ja00";
        const formspreeUrl = "https://formspree.io/f/mnjwqyqd";

        orderForm.addEventListener('submit', async function(e) {
            e.preventDefault();
            loadingOverlay.style.display = 'flex';

            const formData = new FormData(orderForm);

            try {
                const response = await fetch(formspreeUrl, {
                    method: 'POST',
                    body: formData,
                    headers: {
                        'Accept': 'application/json'
                    }
                });

                if (response.ok) {
                    // Erfolgreich gesendet -> Jetzt zu Stripe weiterleiten
                    window.location.href = stripeLink;
                } else {
                    alert('Hoppla! Da gab es ein Problem. Bitte versuche es erneut oder nutze LINE Support.');
                    loadingOverlay.style.display = 'none';
                }
            } catch (error) {
                alert('Netzwerkfehler. Bitte prüfe deine Internetverbindung.');
                loadingOverlay.style.display = 'none';
            }
        });
    </script>
